# GTA VI Hardware & FPS Simulator

Glassmorphic `PySide6` desktop dashboard estimating real-time GTA VI **Average FPS**, **1% Lows**, **Frame Generation output**, **VRAM / System RAM / Unified Pool usage**, and **CPU vs GPU/RT bottlenecks** based on the leaked Tex2 console settings table and 2025/2026 PC hardware profiles.

## Features

- **Live Telemetry (Zero "Calculate" Button):** Every hardware change, toggle, display tweak, or graphics setting updates all gauges, warnings, and multi-GPU charts instantaneously.
- **System Hardware Configurator:**
  - 9 PC GPUs (`RTX 5090` down to `Laptop Arc 140V`) + `PS5` and `PS5 Pro` reference baselines
  - 8 CPUs (`Ryzen 7 9800X3D`, `i5-13400F`, `Core Ultra 7 256V`, etc.)
  - RAM capacity (`16/32/64 GB`), RAM speed profiles (`DDR4-3200` to `DDR5-6000 CL30` & `LPDDR5X-8533`), `Resizable BAR`, `GPU OC (+0% to +7%)`, `NVMe Gen4 vs SATA/HDD`, and `Laptop 20–25W Thermal Coupling`
  - Quick hardware presets (`My Desktop (13400F + B580 + 4800)`, `My Laptop (TP3407SA 256V)`, `High-End (9800X3D + 5090)`)
- **Tex2 Graphics Matrix & Presets:** `Series S`, `PS5 / Series X`, `PS5 Pro`, `900p Sweet Spot`, and `All Ultra` across all 6 categories (`Lighting & RT`, `Textures & World`, `Water`, `Volumetrics & Particles`, `Hair`, `Population / AI / Physics`) with `CPU`/`GPU`/`RT`/`VRAM`/`RAM` impact badges and optional `EN / فارسی` diagnosis & tooltip toggle.
- **Auto-Optimize for 60 FPS + Undo:** Intelligently reduces only the settings causing the active bottleneck (with one-click Undo).
- **Live Multi-GPU Comparison Chart:** Compares Base FPS of all GPUs plus PS5 / PS5 Pro references with 30 FPS & 60 FPS marker lines.

## Quick Start

```powershell
# Using the pre-created virtual environment:
.venv\Scripts\python.exe main.py

# Or standard installation:
pip install -r requirements.txt
python main.py
```

## Run Tests

```powershell
.venv\Scripts\python.exe -m pytest -v
```

def get_platform_compatibility_notes(
    gpu: GPUProfile,
    cpu: CPUProfile,
    ram_speed: RAMSpeedProfile,
    is_laptop_mode: bool,
) -> list[str]:
    """Return informational (non-blocking) hardware pairing hints."""
    notes: list[str] = []
    if cpu.socket_hint == "AM5" and ram_speed.ddr_gen == "DDR4":
        notes.append("Hint: AM5 CPUs require DDR5 memory on real motherboards.")
    elif cpu.socket_hint == "AM4" and ram_speed.ddr_gen != "DDR4":
        notes.append("Hint: AM4 CPUs use DDR4 memory on real motherboards.")
    elif cpu.socket_hint == "LGA1700" and ram_speed.ddr_gen == "LPDDR5X":
        notes.append("Hint: LPDDR5X is soldered mobile memory (Lunar Lake).")

    if is_laptop_mode and (not gpu.is_laptop_thermal or not cpu.is_laptop_thermal):
        notes.append("Laptop thermal throttling (20-25W TDP) is active in Laptop Mode.")
    if gpu.is_unified_memory and ram_speed.ddr_gen != "LPDDR5X":
        notes.append("Hint: Arc 140V relies on on-package LPDDR5X-8533 unified RAM.")
    return notes

from __future__ import annotations

from dataclasses import dataclass
from typing import Literal

BadgeType = Literal["CPU", "GPU", "RT", "VRAM", "RAM"]


@dataclass(frozen=True, slots=True)
class ResolutionProfile:
    """Output display resolution scaling factor and framebuffer VRAM delta."""

    id: str
    label: str
    base_res_factor: float
    vram_delta_gb: float


@dataclass(frozen=True, slots=True)
class UpscalerProfile:
    """Internal render resolution multiplier for native or spatial/temporal upscalers."""

    id: str
    generic_label: str
    render_multiplier: float
    description_en: str
    description_fa: str

    def display_label(self, brand: str) -> str:
        """Return vendor-specific pill label (DLSS / FSR / XeSS)."""
        if self.id == "native":
            return "Native"
        if self.id == "native_aa":
            aa_name = "DLAA" if brand == "DLSS" else f"{brand} AA"
            return f"Native AA ({aa_name})"
        return f"{brand} {self.generic_label}"


@dataclass(frozen=True, slots=True)
class FrameGenProfile:
    """Frame generation mode multiplier and compatibility constraint."""

    id: str
    label: str
    fps_multiplier: float
    base_overhead: float
    requires_fg: bool
    requires_mfg: bool


@dataclass(frozen=True, slots=True)
class SettingOption:
    """Single quality tier for a graphics setting with deltas relative to PS5 baseline."""

    label: str
    raster_cost: float = 0.0
    rt_cost: float = 0.0
    cpu_cost: float = 0.0
    vram_mb: float = 0.0
    ram_mb: float = 0.0


@dataclass(frozen=True, slots=True)
class SettingDefinition:
    """Metadata and options for a categorized GTA VI graphics setting."""

    id: str
    name: str
    category: str
    badges: tuple[BadgeType, ...]
    options: tuple[SettingOption, ...]
    default_option: str
    tex2_subsettings: str
    tooltip_en: str
    tooltip_fa: str

    def option_labels(self) -> list[str]:
        return [opt.label for opt in self.options]

    def get_option(self, label: str) -> SettingOption:
        for opt in self.options:
            if opt.label == label:
                return opt
        return self.get_option_default()

    def get_option_default(self) -> SettingOption:
        for opt in self.options:
            if opt.label == self.default_option:
                return opt
        return self.options[0]

    values: dict[str, str]
    resolution_id: str | None = None

def _damp_cost_above_baseline(cost: float, slope: float = 0.32, floor: float = 0.45) -> float:
    """Preserve <=1.0 savings linearly while modeling engine pass overlap above 1.0."""
    clamped = max(floor, cost)
    if clamped <= 1.0:
        return clamped
    return 1.0 + slope * (clamped - 1.0)


def _smooth_min(a: float, b: float, power: float = 16.0) -> float:
    """Smooth min(a, b) with realistic ~4% frame-time contention when CPU and GPU match."""
    safe_a = max(1.0, a)
    safe_b = max(1.0, b)
    return 1.0 / ((safe_a ** (-power) + safe_b ** (-power)) ** (1.0 / power))


def _resolve_fg(gpu: GPUProfile, requested_fg_id: str) -> tuple[str, float, float, str | None]:
    """Validate frame generation compatibility against the selected GPU."""
    fg = FRAME_GEN_BY_ID.get(requested_fg_id, FRAME_GEN_BY_ID["off"])
    if fg.id == "off":
        return "off", 1.0, 0.0, None
    if not gpu.supports_fg:
        return "off", 1.0, 0.0, f"{gpu.name} does not support PC Frame Generation; reverted to Off."
    if fg.requires_mfg and not gpu.supports_mfg:
        fallback = FRAME_GEN_BY_ID["2x"]
        return (
            fallback.id,
            fallback.fps_multiplier,
            fallback.base_overhead,
            f"MFG ({fg.label}) requires RTX 50-series; fell back to 2x FG ({gpu.fg_label}).",
        )
    return fg.id, fg.fps_multiplier, fg.base_overhead, None


def _accumulate_costs(settings_map: dict[str, str]) -> tuple[float, float, float, float, float]:
    """Sum setting deltas relative to PS5 / Series X baseline."""
    raster = 1.0
    rt = 1.0
    cpu = 1.0
    vram_mb = 0.0
    ram_mb = 0.0

    for definition in SETTINGS:
        chosen = settings_map.get(definition.id, definition.default_option)
        opt = definition.get_option(chosen)
        raster += opt.raster_cost
        rt += opt.rt_cost
        cpu += opt.cpu_cost
        vram_mb += opt.vram_mb
        ram_mb += opt.ram_mb

    return raster, rt, cpu, vram_mb, ram_mb


def simulate(
    sim_input: SimulationInput,
    gpu_override: GPUProfile | None = None,
    cpu_override: CPUProfile | None = None,
) -> SimulationResult:
    """Run deterministic telemetry simulation for the provided hardware and settings."""
    gpu = gpu_override or GPU_BY_ID[sim_input.gpu_id]
    cpu: CPUProfile = cpu_override or CPU_BY_ID[sim_input.cpu_id]
    ram_speed: RAMSpeedProfile = RAM_SPEED_BY_ID[sim_input.ram_speed_id]
    res = RESOLUTION_BY_ID[sim_input.resolution_id]
    upscaler = UPSCALER_BY_ID[sim_input.upscaler_id]

    eff_fg_id, fg_mult, _fg_overhead, fg_note = _resolve_fg(gpu, sim_input.frame_gen_id)
    effective_res_factor = res.base_res_factor * upscaler.render_multiplier

    raw_raster, raw_rt, raw_cpu, delta_vram_mb, delta_ram_mb = _accumulate_costs(sim_input.settings)

    damped_raster = _damp_cost_above_baseline(raw_raster, slope=0.32)
    damped_rt = _damp_cost_above_baseline(raw_rt, slope=0.32)
    damped_cpu = _damp_cost_above_baseline(raw_cpu, slope=0.32)

    if cpu.has_3d_vcache and damped_cpu > 1.0:
        # 96MB 3D V-Cache mitigates Ultra BVH/AI cache thrashing.
        damped_cpu = 1.0 + 0.85 * (damped_cpu - 1.0)

    # 1. VRAM & System RAM demand
    fg_vram_gb = 0.55 if eff_fg_id != "off" else 0.0
    vram_used_gb = max(
        3.0,
        7.4 + res.vram_delta_gb + fg_vram_gb + (delta_vram_mb / 1024.0),
    )
    os_ram_gb = 4.2
    game_ram_gb = max(4.5, 8.2 + (delta_ram_mb / 1024.0))
    ram_used_gb = os_ram_gb + game_ram_gb

    is_unified_pool = gpu.is_unified_memory
    if is_unified_pool:
        unified_pool_total_gb = float(sim_input.ram_gb)
        unified_pool_used_gb = os_ram_gb + game_ram_gb + (vram_used_gb * 0.72)
        vram_total_gb = unified_pool_total_gb
        ram_total_gb = unified_pool_total_gb
        vram_overflow_gb = max(0.0, unified_pool_used_gb - unified_pool_total_gb)
        ram_overflow_gb = vram_overflow_gb
    else:
        unified_pool_total_gb = float(sim_input.ram_gb)
        unified_pool_used_gb = ram_used_gb
        vram_total_gb = gpu.vram_gb
        ram_total_gb = float(sim_input.ram_gb)
        vram_overflow_gb = max(0.0, vram_used_gb - (gpu.vram_gb - 0.3))
        ram_overflow_gb = max(0.0, ram_used_gb - float(sim_input.ram_gb))

    # 2. Raw GPU FPS ceiling (calibrated around PS5 GPU = 100 baseline)
    rt_share = 0.45 * (damped_rt / max(0.5, 0.55 * damped_raster + 0.45 * damped_rt))
    rt_share = min(0.62, max(0.30, rt_share))
    raster_share = 1.0 - rt_share

    weighted_gpu_score = (raster_share * gpu.raster_score + rt_share * gpu.rt_score) / 100.0
    oc_mult = 1.0 + (max(0, min(7, sim_input.gpu_oc_percent)) / 100.0)
    weighted_gpu_demand = raster_share * damped_raster + rt_share * damped_rt

    raw_gpu_fps = (weighted_gpu_score * oc_mult / weighted_gpu_demand) * 32.0 / effective_res_factor

    # Dedicated VRAM or Unified Pool GPU spillover penalty
    if is_unified_pool:
        vram_avg_mult = max(0.50, 1.0 - vram_overflow_gb * 0.10) if vram_overflow_gb > 0 else 1.0
        vram_low_mult = max(0.30, 1.0 - vram_overflow_gb * 0.24) if vram_overflow_gb > 0 else 1.0
    else:
        pcie_penalty = 0.22 if gpu.pcie_lanes == "x8" else 0.16
        vram_avg_mult = max(0.30, 1.0 - vram_overflow_gb * pcie_penalty) if vram_overflow_gb > 0 else 1.0
        vram_low_mult = max(0.20, 1.0 - vram_overflow_gb * 0.32) if vram_overflow_gb > 0 else 1.0

    # 3. Raw CPU FPS ceiling
    if gpu.is_reference_console:
        driver_overhead = 0.0
    else:
        base_overhead = (
            gpu.driver_cpu_overhead if sim_input.rebar_enabled else gpu.driver_cpu_overhead_no_rebar
        )
        driver_overhead = base_overhead * _RES_OVERHEAD_SCALE.get(res.id, 0.90)

    effective_cpu_score = cpu.gaming_score * ram_speed.cpu_multiplier * (1.0 - driver_overhead)
    raw_cpu_fps = (effective_cpu_score * 0.88) / damped_cpu

    # 4. System RAM swap penalty (desktop dedicated or unified pool)
    if ram_overflow_gb > 0 and not is_unified_pool:
        ram_avg_mult = 0.65
        ram_low_mult = 0.40
    elif ram_overflow_gb > 0 and is_unified_pool:
        ram_avg_mult = 0.80
        ram_low_mult = 0.58
    else:
        ram_avg_mult = 1.0
        ram_low_mult = 1.0

    # 5. Laptop thermal & power coupling penalty (20-25W shared TDP)
    is_thermal_coupled = (
        (sim_input.is_laptop_mode or gpu.is_laptop_thermal or cpu.is_laptop_thermal)
        and not gpu.is_reference_console
    )
    thermal_mult = 0.82 if is_thermal_coupled else 1.0

    effective_gpu_fps = raw_gpu_fps * vram_avg_mult * thermal_mult
    effective_cpu_fps = raw_cpu_fps * ram_avg_mult * thermal_mult

    # 6. Smooth bottleneck blending & 1% Low FPS
    base_fps = _smooth_min(effective_gpu_fps, effective_cpu_fps, power=16.0)

    # 1% Low calculation
    cpu_low_ram_ratio = ram_speed.low_1pct_multiplier / ram_speed.cpu_multiplier
    cpu_low_fps = effective_cpu_fps * 0.785 * cpu_low_ram_ratio
    gpu_low_fps = effective_gpu_fps * 0.86

    low_1pct_fps = _smooth_min(cpu_low_fps, gpu_low_fps, power=12.0)
    low_1pct_fps *= vram_low_mult
    if not is_unified_pool:
        low_1pct_fps *= ram_low_mult
    if sim_input.storage_type == "SATA / HDD":
        low_1pct_fps *= 0.72
    low_1pct_fps = min(base_fps * 0.94, max(1.0, low_1pct_fps))

    # Frame generation output
    fg_fps = base_fps * fg_mult if eff_fg_id != "off" else base_fps

    # Component Load % estimates
    gpu_load_pct = min(100.0, max(12.0, (base_fps / max(1.0, effective_gpu_fps)) * 99.0))
    rt_load_pct = min(100.0, max(10.0, gpu_load_pct * min(1.12, 0.78 * damped_rt)))
    cpu_load_pct = min(100.0, max(10.0, (base_fps / max(1.0, effective_cpu_fps)) * 96.0))

    # Determine primary bottleneck and warnings
    warnings: list[WarningMessage] = []
    if vram_overflow_gb > 0 and not is_unified_pool:
        primary_bottleneck: BottleneckKind = "VRAM Overflow"
    elif ram_overflow_gb > 0:
        primary_bottleneck = "RAM / Unified Overflow"
    elif abs(effective_gpu_fps - effective_cpu_fps) / max(effective_gpu_fps, effective_cpu_fps) <= 0.08:
        primary_bottleneck = "Balanced"
    elif effective_gpu_fps < effective_cpu_fps:
        primary_bottleneck = "GPU / RT"
    else:
        primary_bottleneck = "CPU"

    if not gpu.is_reference_console:
        warnings.extend(
            _build_warnings(
                sim_input=sim_input,
                gpu=gpu,
                cpu=cpu,
                ram_speed=ram_speed,
                base_fps=base_fps,
                eff_fg_id=eff_fg_id,
                fg_note=fg_note,
                vram_used_gb=vram_used_gb,
                vram_total_gb=vram_total_gb,
                vram_overflow_gb=vram_overflow_gb,
                ram_used_gb=ram_used_gb,
                ram_total_gb=ram_total_gb,
                ram_overflow_gb=ram_overflow_gb,
                is_unified_pool=is_unified_pool,
                unified_pool_used_gb=unified_pool_used_gb,
                unified_pool_total_gb=unified_pool_total_gb,
            )
        )

    diag_en, diag_fa, rec_en, rec_fa = _build_diagnosis(
        primary_bottleneck=primary_bottleneck,
        gpu=gpu,
        cpu=cpu,
        ram_speed=ram_speed,
        rebar_enabled=sim_input.rebar_enabled,
        effective_gpu_fps=effective_gpu_fps,
        effective_cpu_fps=effective_cpu_fps,
        vram_used_gb=vram_used_gb,
        vram_total_gb=vram_total_gb,
        vram_overflow_gb=vram_overflow_gb,
        ram_overflow_gb=ram_overflow_gb,
        is_unified_pool=is_unified_pool,
        unified_pool_used_gb=unified_pool_used_gb,
        unified_pool_total_gb=unified_pool_total_gb,
        damped_rt=damped_rt,
        damped_raster=damped_raster,
        is_thermal_coupled=is_thermal_coupled,
        storage_type=sim_input.storage_type,
    )

    return SimulationResult(
        base_fps=round(base_fps, 1),
        low_1pct_fps=round(low_1pct_fps, 1),
        fg_fps=round(fg_fps, 1),
        effective_fg_id=eff_fg_id,
        fg_fallback_note=fg_note,
        raw_gpu_fps=round(effective_gpu_fps, 1),
        raw_cpu_fps=round(effective_cpu_fps, 1),
        gpu_load_pct=round(gpu_load_pct, 1),
        rt_load_pct=round(rt_load_pct, 1),
        cpu_load_pct=round(cpu_load_pct, 1),
        vram_used_gb=round(vram_used_gb, 2),
        vram_total_gb=round(vram_total_gb, 1),
        vram_overflow_gb=round(vram_overflow_gb, 2),
        ram_used_gb=round(ram_used_gb, 2),
        ram_total_gb=round(ram_total_gb, 1),
        ram_overflow_gb=round(ram_overflow_gb, 2),
        is_unified_pool=is_unified_pool,
        unified_pool_used_gb=round(unified_pool_used_gb, 2),
        unified_pool_total_gb=round(unified_pool_total_gb, 1),
        effective_res_factor=round(effective_res_factor, 3),
        total_raster_cost=round(raw_raster, 2),
        total_rt_cost=round(raw_rt, 2),
        total_cpu_cost=round(raw_cpu, 2),
        primary_bottleneck=primary_bottleneck,
        diagnosis_en=diag_en,
        diagnosis_fa=diag_fa,
        recommendation_en=rec_en,
        recommendation_fa=rec_fa,
        warnings=tuple(warnings),
    )


def _build_warnings(
    *,
    sim_input: SimulationInput,
    gpu: GPUProfile,
    cpu: CPUProfile,
    ram_speed: RAMSpeedProfile,
    base_fps: float,
    eff_fg_id: str,
    fg_note: str | None,
    vram_used_gb: float,
    vram_total_gb: float,
    vram_overflow_gb: float,
    ram_used_gb: float,
    ram_total_gb: float,
    ram_overflow_gb: float,
    is_unified_pool: bool,
    unified_pool_used_gb: float,
    unified_pool_total_gb: float,
) -> list[WarningMessage]:
    warnings: list[WarningMessage] = []

    if is_unified_pool:
        ratio = unified_pool_used_gb / max(1.0, unified_pool_total_gb)
        if vram_overflow_gb > 0:
            warnings.append(
                WarningMessage(
                    severity="critical",
                    code="unified_overflow",
                    title_en=f"Unified Memory Overflow (+{vram_overflow_gb:.1f} GB)",
                    detail_en=(
                        f"Combined OS + GTA VI + VRAM pool needs {unified_pool_used_gb:.1f} GB on a "
                        f"{unified_pool_total_gb:.0f} GB shared pool. Expect pagefile thrashing & heavy stutters."
                    ),
                    title_fa=f"سرریز حافظه یکپارچه لپ‌تاپ (+{vram_overflow_gb:.1f} گیگابایت)",
                    detail_fa=(
                        f"مجموع مصرف سیستم‌عامل، بازی و VRAM به {unified_pool_used_gb:.1f} گیگابایت رسیده "
                        f"که از کل ظرفیت {unified_pool_total_gb:.0f} گیگابایتی بیشتر است و باعث لگ شدید می‌شود."
                    ),
                )
            )
        elif ratio >= 0.88:
            warnings.append(
                WarningMessage(
                    severity="warning",
                    code="unified_tight",
                    title_en=f"Tight Unified Pool ({unified_pool_used_gb:.1f} / {unified_pool_total_gb:.0f} GB)",
                    detail_en="Shared memory is above 88% utilization; background apps may trigger stutter.",
                    title_fa=f"فشار بالای حافظه یکپارچه ({unified_pool_used_gb:.1f} از {unified_pool_total_gb:.0f} گیگابایت)",
                    detail_fa="بیش از ۸۸٪ حافظه مشترک پر شده است؛ برنامه‌های پس‌زمینه می‌توانند افت فریم بدهند.",
                )
            )
    else:
        if vram_overflow_gb > 0:
            bus_note = " over PCIe x8!" if gpu.pcie_lanes == "x8" else "."
            warnings.append(
                WarningMessage(
                    severity="critical",
                    code="vram_overflow",
                    title_en=f"{vram_total_gb:.0f}GB VRAM Overflow (+{vram_overflow_gb:.1f} GB)",
                    detail_en=(
                        f"Requested {vram_used_gb:.1f} GB exceeds {vram_total_gb:.0f} GB VRAM, spilling "
                        f"into system RAM{bus_note} Severe 1% Low collapse."
                    ),
                    title_fa=f"سرریز حافظه VRAM کارت {vram_total_gb:.0f} گیگابایتی (+{vram_overflow_gb:.1f} GB)",
                    detail_fa=(
                        f"مصرف {vram_used_gb:.1f} گیگابایت از ظرفیت {vram_total_gb:.0f} گیگابایت کارت عبور کرده "
                        "و باعث افت شدید فریم لحظه‌ای (1% Low) می‌شود."
                    ),
                )
            )
        elif (vram_used_gb / max(1.0, vram_total_gb)) >= 0.88:
            warnings.append(
                WarningMessage(
                    severity="warning",
                    code="vram_near_limit",
                    title_en=f"High VRAM Pressure ({vram_used_gb:.1f} / {vram_total_gb:.0f} GB)",
                    detail_en="VRAM usage exceeds 88%; dense downtown scenes may spike textures.",
                    title_fa=f"مصرف بالای VRAM ({vram_used_gb:.1f} از {vram_total_gb:.0f} گیگابایت)",
                    detail_fa="مصرف VRAM بالای ۸۸٪ است و در مناطق شلوغ شهر ممکن است نوسان ایجاد کند.",
                )
            )

        if ram_overflow_gb > 0:
            warnings.append(
                WarningMessage(
                    severity="critical",
                    code="ram_overflow",
                    title_en=f"System RAM Overflow (+{ram_overflow_gb:.1f} GB)",
                    detail_en=(
                        f"{ram_used_gb:.1f} GB required on {ram_total_gb:.0f} GB RAM — Windows pagefile "
                        "swapping cuts 1% Lows dramatically."
                    ),
                    title_fa=f"کمبود رم سیستم (+{ram_overflow_gb:.1f} گیگابایت)",
                    detail_fa=(
                        f"مصرف {ram_used_gb:.1f} گیگابایت از رم {ram_total_gb:.0f} گیگابایتی بیشتر است "
                        "و Swap دیسک باعث لگ شدید می‌شود."
                    ),
                )
            )

    if not sim_input.rebar_enabled and gpu.id == "arc_b580":
        warnings.append(
            WarningMessage(
                severity="critical",
                code="arc_no_rebar",
                title_en="Resizable BAR OFF on Intel Arc B580",
                detail_en="Intel Arc driver CPU overhead jumps from 7% to 28% without ReBAR!",
                title_fa="خاموش بودن Resizable BAR روی کارت Intel Arc B580",
                detail_fa="بدون ReBAR سربار پردازنده درایور اینتل از ۷٪ به ۲۸٪ جهش می‌کند!",
            )
        )

    if eff_fg_id != "off" and base_fps < 38.0:
        warnings.append(
            WarningMessage(
                severity="warning",
                code="fg_low_base",
                title_en="High Input Latency / Artifacts Risk (Base FPS < 38)",
                detail_en=(
                    f"Base FPS is {base_fps:.1f}; enabling Frame Generation below 38 FPS causes sluggish "
                    "controls and HUD/motion artifacts."
                ),
                title_fa="تاخیر ورودی بالا / ریسک آرتیفکت تصویر (فریم پایه زیر ۳۸)",
                detail_fa=(
                    f"فریم پایه {base_fps:.1f} است؛ فعال‌سازی Frame Gen زیر ۳۸ فریم باعث کندی کنترل و خطای تصویر می‌شود."
                ),
            )
        )

    if sim_input.storage_type == "SATA / HDD":
        warnings.append(
            WarningMessage(
                severity="warning",
                code="slow_storage",
                title_en="SATA / HDD Streaming Penalty",
                detail_en="High-speed driving streams RAGE 9 assets too slowly on SATA/HDD (-28% 1% Lows).",
                title_fa="جریمه سرعت دیسک SATA / HDD",
                detail_fa="بارگذاری سریع بافت‌ها هنگام رانندگی روی SATA/HDD با افت ۲۸٪ در 1% Low همراه است.",
            )
        )

    if fg_note:
        warnings.append(
            WarningMessage(
                severity="info",
                code="fg_fallback",
                title_en="Frame Gen Adjusted",
                detail_en=fg_note,
                title_fa="تنظیم خودکار Frame Gen",
                detail_fa=fg_note,
            )
        )

    for note in get_platform_compatibility_notes(gpu, cpu, ram_speed, sim_input.is_laptop_mode):
        warnings.append(
            WarningMessage(
                severity="info",
                code="platform_hint",
                title_en="Platform Note",
                detail_en=note,
                title_fa="نکته سخت‌افزاری",
                detail_fa=note,
            )
        )

    return warnings


def _build_diagnosis(
    *,
    primary_bottleneck: BottleneckKind,
    gpu: GPUProfile,
    cpu: CPUProfile,
    ram_speed: RAMSpeedProfile,
    rebar_enabled: bool,
    effective_gpu_fps: float,
    effective_cpu_fps: float,
    vram_used_gb: float,
    vram_total_gb: float,
    vram_overflow_gb: float,
    ram_overflow_gb: float,
    is_unified_pool: bool,
    unified_pool_used_gb: float,
    unified_pool_total_gb: float,
    damped_rt: float,
    damped_raster: float,
    is_thermal_coupled: bool,
    storage_type: StorageType,
) -> tuple[str, str, str, str]:
    thermal_suffix_en = (
        " Shared 20-25W laptop TDP also caps sustained boost." if is_thermal_coupled else ""
    )
    thermal_suffix_fa = (
        " محدودیت توان مشترک ۲۰ تا ۲۵ واتی لپ‌تاپ نیز سقف کلاک را پایین می‌آورد."
        if is_thermal_coupled
        else ""
    )

    if primary_bottleneck == "VRAM Overflow":
        diag_en = (
            f"VRAM Spillover Bottleneck: GTA VI requests {vram_used_gb:.1f} GB on {gpu.name}'s "
            f"{vram_total_gb:.0f} GB buffer (+{vram_overflow_gb:.1f} GB spill over {gpu.pcie_lanes})."
        )
        diag_fa = (
            f"گلوگاه سرریز VRAM: بازی به {vram_used_gb:.1f} گیگابایت نیاز دارد در حالی که کارت "
            f"{vram_total_gb:.0f} گیگابایت حافظه دارد (+{vram_overflow_gb:.1f} GB سرریز)."
        )
        rec_en = (
            "Drop Texture Quality to High/Medium, Vehicle HD Count to High, or disable RT Shadows "
            "to keep allocation under the VRAM cap."
        )
        rec_fa = (
            "کیفیت بافت (Texture) و بافت خودروها را یک پله کم کنید یا RT Shadows را ببندید "
            "تا مصرف به زیر سقف VRAM برگردد."
        )
        return diag_en, diag_fa, rec_en, rec_fa

    if primary_bottleneck == "RAM / Unified Overflow":
        if is_unified_pool:
            diag_en = (
                f"Unified Memory Pool Bottleneck: Shared OS + CPU + GPU demand is "
                f"{unified_pool_used_gb:.1f} GB vs {unified_pool_total_gb:.0f} GB LPDDR5X.{thermal_suffix_en}"
            )
            diag_fa = (
                f"گلوگاه حافظه یکپارچه لپ‌تاپ: مجموع مصرف سیستم و گرافیک به {unified_pool_used_gb:.1f} از "
                f"{unified_pool_total_gb:.0f} گیگابایت رسیده است.{thermal_suffix_fa}"
            )
            rec_en = (
                "Lower Resolution to 720p, Texture Quality to Medium, and Pedestrian Density to Medium/Low."
            )
            rec_fa = (
                "رزولوشن را روی 720p بگذارید و Texture Quality و تراکم جمعیت را روی Medium/Low قرار دهید."
            )
        else:
            diag_en = (
                f"System RAM Overflow: {ram_overflow_gb:.1f} GB beyond physical RAM forces Windows "
                "disk paging during streaming."
            )
            diag_fa = (
                f"گلوگاه کمبود رم سیستم: {ram_overflow_gb:.1f} گیگابایت سرریز به دیسک باعث افت شدید فریم می‌شود."
            )
            rec_en = "Drop Pedestrian & Vehicle Density to Medium/Low and close background browsers."
            rec_fa = "تراکم عابران و خودروها را کاهش دهید و برنامه‌های پس‌زمینه را ببندید."
        return diag_en, diag_fa, rec_en, rec_fa

    if primary_bottleneck == "CPU":
        overhead_txt = (
            " + Arc ReBAR-OFF overhead (28%)"
            if (gpu.id == "arc_b580" and not rebar_enabled)
            else ""
        )
        diag_en = (
            f"CPU Simulation Bottleneck: {cpu.name} + {ram_speed.name}{overhead_txt} caps FPS around "
            f"{effective_cpu_fps:.0f} FPS while GPU headroom reaches {effective_gpu_fps:.0f} FPS."
            f"{thermal_suffix_en}"
        )
        diag_fa = (
            f"گلوگاه پردازنده (CPU): ترکیب {cpu.name} و {ram_speed.name} سقف فریم را روی حدود "
            f"{effective_cpu_fps:.0f} فریم محدود کرده، در حالی که کارت گرافیک توان {effective_gpu_fps:.0f} فریم را دارد."
            f"{thermal_suffix_fa}"
        )
        if gpu.id == "arc_b580" and not rebar_enabled:
            rec_en = "Enable Resizable BAR in BIOS immediately, then reduce Pedestrian Density or BVH Quality."
            rec_fa = "حتماً Resizable BAR را در بایوس روشن کنید و سپس تراکم جمعیت یا BVH Quality را کم کنید."
        else:

def compare_all_gpus(sim_input: SimulationInput) -> list[GPUComparisonEntry]:
    """Compute Base FPS and 1% Low across all 9 GPUs + PS5 / PS5 Pro console references."""
    entries: list[GPUComparisonEntry] = []

    for gpu in GPUS:
        # Compare each GPU fairly on the user's CPU/RAM/settings; if comparing Arc 140V from desktop,
        # simulate its coupled 16GB shared pool and thermal envelope accurately.
        if gpu.id == sim_input.gpu_id:
            res = simulate(sim_input)
        elif gpu.is_unified_memory:
            laptop_input = SimulationInput(
                gpu_id=gpu.id,
                cpu_id="ultra_7_256v" if sim_input.is_laptop_mode else sim_input.cpu_id,
                ram_gb=16,
                ram_speed_id="lpddr5x_8533",
                rebar_enabled=True,
                gpu_oc_percent=0,
                storage_type=sim_input.storage_type,
                is_laptop_mode=True,
                resolution_id=sim_input.resolution_id,
                upscaler_id=sim_input.upscaler_id,
                frame_gen_id="off",
                settings=sim_input.settings,
            )
            res = simulate(laptop_input, gpu_override=gpu)
        else:
            res = simulate(sim_input, gpu_override=gpu)

        entries.append(
            GPUComparisonEntry(
                gpu_id=gpu.id,
                label=gpu.short_name,
                base_fps=res.base_fps,
                low_1pct_fps=res.low_1pct_fps,
                is_selected=(gpu.id == sim_input.gpu_id),
                is_reference_console=False,
                has_vram_overflow=(res.vram_overflow_gb > 0),
                notes=gpu.notes,
            )
        )

    for console in CONSOLE_REFERENCES:
        console_cpu = CPUProfile(
            id=f"{console.id}_cpu",
            name="Console Zen 2",
            gaming_score=58.0 if console.id == "ps5_ref" else 63.0,
            cores_threads="8C / 16T",
            architecture_notes="Custom AMD Zen 2 Console APU",
            socket_hint="Console",
        )
        console_input = SimulationInput(
            gpu_id=console.id,
            cpu_id="ryzen_5600x",
            ram_gb=16,
            ram_speed_id="ddr5_5600",
            rebar_enabled=True,
            gpu_oc_percent=0,
            storage_type="NVMe Gen4",
            is_laptop_mode=False,
            resolution_id=sim_input.resolution_id,
            upscaler_id=sim_input.upscaler_id,
            frame_gen_id="off",
            settings=sim_input.settings,
        )
        res = simulate(console_input, gpu_override=console, cpu_override=console_cpu)
        entries.append(
            GPUComparisonEntry(
                gpu_id=console.id,
                label=console.short_name,
                base_fps=res.base_fps,
                low_1pct_fps=res.low_1pct_fps,
                is_selected=False,
                is_reference_console=True,
                has_vram_overflow=(res.vram_overflow_gb > 0),
                notes=console.notes,
            )
        )

    entries.sort(key=lambda item: item.base_fps, reverse=True)
    return entries

def _step_down_setting(settings_map: dict[str, str], setting_id: str) -> str | None:
    """Lower one setting by one quality tier; return new label or None if already minimum."""
    definition = SETTING_BY_ID[setting_id]
    current = settings_map.get(setting_id, definition.default_option)
    labels = definition.option_labels()
    if current not in labels:
        return None
    idx = labels.index(current)
    if idx <= 0:
        return None
    new_label = labels[idx - 1]
    settings_map[setting_id] = new_label
    return new_label


def auto_optimize_for_60_fps(sim_input: SimulationInput, target_fps: float = 60.0) -> AutoOptimizeResult:
    """Intelligently lower only settings causing the active bottleneck toward 60 Base FPS."""
    working_settings = dict(sim_input.settings)
    working_upscaler = sim_input.upscaler_id
    changed: list[str] = []

    def _current_sim() -> SimulationResult:
        return simulate(
            SimulationInput(
                gpu_id=sim_input.gpu_id,
                cpu_id=sim_input.cpu_id,
                ram_gb=sim_input.ram_gb,
                ram_speed_id=sim_input.ram_speed_id,
                rebar_enabled=sim_input.rebar_enabled,
                gpu_oc_percent=sim_input.gpu_oc_percent,
                storage_type=sim_input.storage_type,
                is_laptop_mode=sim_input.is_laptop_mode,
                resolution_id=sim_input.resolution_id,
                upscaler_id=working_upscaler,
                frame_gen_id=sim_input.frame_gen_id,
                settings=working_settings,
            )
        )

    res = _current_sim()

    # Step 1: Clear VRAM or RAM/Unified overflow first.
    for _ in range(24):
        if res.vram_overflow_gb <= 0 and res.ram_overflow_gb <= 0:
            break
        order = (
            ("ped_vehicle_density",) + _VRAM_PRIORITY
            if res.ram_overflow_gb > 0
            else _VRAM_PRIORITY
        )
        progressed = False
        for sid in order:
            old_val = working_settings.get(sid, SETTING_BY_ID[sid].default_option)
            new_val = _step_down_setting(working_settings, sid)
            if new_val is not None:
                changed.append(f"{SETTING_BY_ID[sid].name}: {old_val} -> {new_val}")
                progressed = True
                break
        if not progressed:
            break
        res = _current_sim()

    # Step 2: Iteratively trim settings targeting whichever ceiling is limiting FPS.
    for _ in range(40):
        if res.base_fps >= target_fps:
            break
        if res.raw_cpu_fps <= res.raw_gpu_fps * 1.05:
            priority = _CPU_PRIORITY
        else:
            priority = _GPU_PRIORITY

        progressed = False
        for sid in priority:
            old_val = working_settings.get(sid, SETTING_BY_ID[sid].default_option)
            new_val = _step_down_setting(working_settings, sid)
            if new_val is not None:
                changed.append(f"{SETTING_BY_ID[sid].name}: {old_val} -> {new_val}")
                progressed = True
                break
        if not progressed:
            break
        res = _current_sim()

    # Step 3: If GPU is still capping below 60 FPS, step Upscaler up to Balanced (never touch Res/FG).
    while res.base_fps < target_fps and res.raw_gpu_fps < res.raw_cpu_fps * 1.15:
        if working_upscaler in ("balanced", "performance"):
            break
        if working_upscaler in ("native", "native_aa"):
            next_upscaler = "quality"
        elif working_upscaler == "quality":
            next_upscaler = "balanced"
        else:
            break
        changed.append(
            f"Upscaler: {UPSCALER_BY_ID[working_upscaler].generic_label} -> "
            f"{UPSCALER_BY_ID[next_upscaler].generic_label}"
        )
        working_upscaler = next_upscaler
        res = _current_sim()

    reached = res.base_fps >= target_fps
    if not changed and reached:
        summary_en = f"Already achieving {res.base_fps:.1f} Base FPS — no downgrades needed."
        summary_fa = f"در حال حاضر به {res.base_fps:.1f} فریم پایه رسیده‌اید؛ نیازی به کاهش تنظیمات نیست."
    elif reached:
        summary_en = (
            f"Reached {res.base_fps:.1f} Base FPS with {len(changed)} targeted adjustment(s)."
        )
        summary_fa = (
            f"با {len(changed)} تغییر هدفمند روی گلوگاه اصلی به {res.base_fps:.1f} فریم پایه رسیدید."
        )
    else:
        summary_en = (
            f"Maximized to {res.base_fps:.1f} Base FPS ({len(changed)} adjustment(s)); "
            f"hardware/resolution cap prevents 60 Base FPS without lowering Resolution or enabling FG."
        )
        summary_fa = (
            f"بیشینه فریم به {res.base_fps:.1f} فریم پایه رسید ({len(changed)} تغییر)؛ "
            "سقف سخت‌افزار/رزولوشن اجازه ۶۰ فریم پایه بدون کاهش رزولوشن یا Frame Gen را نمی‌دهد."
        )

    return AutoOptimizeResult(
        settings=working_settings,
        upscaler_id=working_upscaler,
        changed_items=tuple(changed),
        reached_target=reached,
        resulting_base_fps=res.base_fps,
        summary_en=summary_en,
        summary_fa=summary_fa,
    )

def test_main_window_presets_and_live_updates(qapp: QApplication) -> None:
    window = MainWindow()
    window.resize(1440, 880)
    window.show()
    qapp.processEvents()

    # Initial Desktop preset -> ~50-56 Base FPS
    initial_fps = float(window._hero_card._avg_value.text())
    assert 48.0 <= initial_fps <= 58.0

    # Switch to High-End (9800X3D + 5090) and verify MFG 4x unlocks
    window._apply_hardware_preset("high_end")
    qapp.processEvents()
    assert window._fg_pills._buttons["4x"].isEnabled() is True
    window._fg_pills.set_selected("4x")
    window._fg_pills.changed.emit("4x")
    qapp.processEvents()
    assert float(window._hero_card._fg_value.text()) >= 200.0

    # Switch back to Laptop preset -> MFG 4x auto-falls back and disables
    window._apply_hardware_preset("my_laptop")
    qapp.processEvents()
    assert window._fg_pills._buttons["4x"].isEnabled() is False
    assert window._fg_pills.current_id() == "2x"
    assert "Unified" in window._vram_bar._title

    # Desktop preset + Auto-Optimize -> reaches >= 60 Base FPS and enables Undo
    window._apply_hardware_preset("my_desktop")
    window._apply_settings_preset("ps5_xsx")
    window._fg_pills.set_selected("off")
    window._fg_pills.changed.emit("off")
    qapp.processEvents()
    window._on_auto_optimize()
    qapp.processEvents()
    assert float(window._hero_card._avg_value.text()) >= 60.0
    assert window._undo_btn.isVisible() is True

class ViceCityBackdropWidget(QWidget):
    """Root container painting the deep Vice City dark gradient & ambient neon glows."""

    def paintEvent(self, event: QPaintEvent) -> None:
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)

        rect = self.rect()
        base_grad = QLinearGradient(0, 0, rect.width(), rect.height())
        base_grad.setColorAt(0.0, QColor(BG_TOP))
        base_grad.setColorAt(0.55, QColor("#111022"))
        base_grad.setColorAt(1.0, QColor(BG_BOTTOM))
        painter.fillRect(rect, base_grad)

        # Ambient Vice City neon cyan glow (top-left)
        cyan_glow = QRadialGradient(QPointF(rect.width() * 0.12, rect.height() * 0.14), rect.width() * 0.42)
        cyan_glow.setColorAt(0.0, QColor(0, 240, 255, 36))
        cyan_glow.setColorAt(1.0, QColor(0, 240, 255, 0))
        painter.fillRect(rect, cyan_glow)

        # Ambient Vice City neon magenta glow (bottom-right)
        mag_glow = QRadialGradient(QPointF(rect.width() * 0.88, rect.height() * 0.86), rect.width() * 0.46)
        mag_glow.setColorAt(0.0, QColor(255, 46, 147, 34))
        mag_glow.setColorAt(1.0, QColor(255, 46, 147, 0))
        painter.fillRect(rect, mag_glow)

        super().paintEvent(event)

    def _apply_hardware_preset(self, preset_id: str, recalculate: bool = True) -> None:
        preset = PRESET_BY_ID[preset_id]
        self._suppress_recalc = True
        try:
            self._set_combo_by_data(self._gpu_combo, preset.gpu_id)
            self._set_combo_by_data(self._cpu_combo, preset.cpu_id)
            self._ram_cap_pills.set_selected(str(preset.ram_gb))
            self._set_combo_by_data(self._ram_speed_combo, preset.ram_speed_id)
            self._rebar_pills.set_selected("on" if preset.rebar_enabled else "off")
            self._oc_slider.setValue(preset.gpu_oc_percent)
            self._oc_val_label.setText(f"+{preset.gpu_oc_percent}%")
            self._storage_pills.set_selected(preset.storage_type)
            self._laptop_pills.set_selected("laptop" if preset.is_laptop_mode else "desktop")
            self._sync_gpu_tech_labels()
            self._highlight_hw_preset(preset_id)
        finally:
            self._suppress_recalc = False
        if recalculate:
            self._recalculate()

    def _apply_settings_preset(self, preset_id: str, recalculate: bool = True) -> None:
        preset = SETTINGS_PRESET_BY_ID[preset_id]
        self._suppress_recalc = True
        try:
            self._settings_state = dict(preset.values)
            for sid, val in self._settings_state.items():
                if sid in self._setting_rows:
                    self._setting_rows[sid].set_value(val)
            if preset.resolution_id:
                self._res_pills.set_selected(preset.resolution_id)
            if preset.upscaler_id:
                self._upscaler_pills.set_selected(preset.upscaler_id)
            self._undo_snapshot = None
            self._undo_btn.setVisible(False)
            self._opt_status_lbl.setVisible(False)
            self._highlight_settings_preset(preset_id)
        finally:
            self._suppress_recalc = False
        if recalculate:
            self._recalculate()

    def _on_gpu_changed(self) -> None:
        if not self._suppress_recalc:
            self._highlight_hw_preset(None)
            self._sync_gpu_tech_labels()
            self._recalculate()

    def _on_hardware_changed(self) -> None:
        if not self._suppress_recalc:
            self._highlight_hw_preset(None)
            self._recalculate()

    def _on_oc_changed(self, value: int) -> None:
        self._oc_val_label.setText(f"+{value}%")
        if not self._suppress_recalc:
            self._highlight_hw_preset(None)
            self._recalculate()

    def _on_display_or_setting_changed(self) -> None:
        if not self._suppress_recalc:
            self._highlight_settings_preset(None)
            self._recalculate()

    def _on_setting_row_changed(self, setting_id: str, new_value: str) -> None:
        self._settings_state[setting_id] = new_value
        if not self._suppress_recalc:
            self._highlight_settings_preset(None)
            self._recalculate()

    def _on_auto_optimize(self) -> None:
        current = self._collect_input()
        self._undo_snapshot = (dict(self._settings_state), current.upscaler_id)

        opt = auto_optimize_for_60_fps(current, target_fps=60.0)
        self._suppress_recalc = True
        try:
            self._settings_state = dict(opt.settings)
            for sid, val in self._settings_state.items():
                if sid in self._setting_rows:
                    self._setting_rows[sid].set_value(val)
            self._upscaler_pills.set_selected(opt.upscaler_id)
            self._highlight_settings_preset(None)
        finally:
            self._suppress_recalc = False

        self._undo_btn.setVisible(len(opt.changed_items) > 0)
        msg = opt.summary_fa if self._lang == "fa" else opt.summary_en
        if opt.changed_items:
            preview = "; ".join(opt.changed_items[:4])
            if len(opt.changed_items) > 4:
                preview += f" (+{len(opt.changed_items) - 4} more)"
            msg = f"{msg}\nChanges: {preview}"
        self._opt_status_lbl.setText(msg)
        self._opt_status_lbl.setVisible(True)
        self._recalculate()

    def _on_undo_optimize(self) -> None:
        if not self._undo_snapshot:
            return
        prev_settings, prev_upscaler = self._undo_snapshot
        self._undo_snapshot = None
        self._suppress_recalc = True
        try:
            self._settings_state = dict(prev_settings)
            for sid, val in self._settings_state.items():
                if sid in self._setting_rows:
                    self._setting_rows[sid].set_value(val)
            self._upscaler_pills.set_selected(prev_upscaler)
        finally:
            self._suppress_recalc = False
        self._undo_btn.setVisible(False)
        self._opt_status_lbl.setVisible(False)
        self._recalculate()

    def _sync_gpu_tech_labels(self) -> None:
        gpu_id = self._gpu_combo.currentData()
        gpu = GPU_BY_ID[gpu_id]

        for u in UPSCALERS:
            self._upscaler_pills.update_label(u.id, u.display_label(gpu.upscaler_brand))

        # Disable MFG 3x/4x on GPUs without RTX 50-series Multi-Frame Gen
        mfg_tip = "" if gpu.supports_mfg else f"Multi-Frame Gen requires RTX 50-series ({gpu.name} supports up to 2x FG)."
        self._fg_pills.set_option_enabled("3x", gpu.supports_mfg, mfg_tip)
        self._fg_pills.set_option_enabled("4x", gpu.supports_mfg, mfg_tip)
        self._fg_pills.set_option_enabled("2x", gpu.supports_fg, "")

        cur_fg = self._fg_pills.current_id()
        if cur_fg in ("3x", "4x") and not gpu.supports_mfg:
            self._fg_pills.set_selected("2x")

    def _collect_input(self) -> SimulationInput:
        storage_str = self._storage_pills.current_id()
        storage: StorageType = "SATA / HDD" if storage_str == "SATA / HDD" else "NVMe Gen4"

        )

def apply_windows_backdrop(widget: QWidget) -> None:
    """Enable Windows 11 dark titlebar & Mica/Acrylic backdrop when supported."""
    if sys.platform != "win32":
        return
    try:
        hwnd = int(widget.winId())
        dwmapi = ctypes.windll.dwmapi
        # DWMWA_USE_IMMERSIVE_DARK_MODE (20)
        dark_mode = ctypes.c_int(1)
        dwmapi.DwmSetWindowAttribute(
            hwnd,
            20,
            ctypes.byref(dark_mode),
            ctypes.sizeof(dark_mode),
        )
        # DWMWA_CAPTION_COLOR (35): match dark Vice City header (#0B0E17 -> COLORREF 0x00170E0B)
        caption_color = ctypes.c_int(0x00170E0B)
        dwmapi.DwmSetWindowAttribute(
            hwnd,
            35,
            ctypes.byref(caption_color),
            ctypes.sizeof(caption_color),
        )
        # DWMWA_SYSTEMBACKDROP_TYPE (38): 2 = Mica, 3 = Acrylic
        backdrop_type = ctypes.c_int(2)
        dwmapi.DwmSetWindowAttribute(
            hwnd,
            38,
            ctypes.byref(backdrop_type),
            ctypes.sizeof(backdrop_type),
        )
    except (OSError, AttributeError, ValueError) as exc:
        # Safe fallback on Windows 10, older builds, or headless/offscreen runs.
        logger.debug("DWM backdrop unavailable: %s", exc)


def fps_accent_color(fps: float) -> str:
    """Return Vice City status color for a given FPS threshold."""
    if fps >= 60.0:

class SegmentedPills(QWidget):
    """Row or grid of pill buttons acting as a sleek mutually-exclusive selector."""

    changed = Signal(str)

    def __init__(
        self,
        options: Iterable[tuple[str, str]],
        selected_id: str | None = None,
        max_cols: int | None = None,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self._max_cols = max_cols
        self._layout = QGridLayout(self)
        self._layout.setContentsMargins(0, 0, 0, 0)
        self._layout.setSpacing(4)

        self._group = QButtonGroup(self)
        self._group.setExclusive(True)
        self._buttons: dict[str, QPushButton] = {}
        self._id_order: list[str] = []

        self.set_options(options, selected_id)

    def set_options(
        self,
        options: Iterable[tuple[str, str]],
        selected_id: str | None = None,
    ) -> None:
        for btn in self._buttons.values():
            self._group.removeButton(btn)
            self._layout.removeWidget(btn)
            btn.deleteLater()
        self._buttons.clear()
        self._id_order.clear()

        opts_list = list(options)
        cols = self._max_cols or max(1, len(opts_list))

        for idx, (opt_id, label) in enumerate(opts_list):
            btn = QPushButton(label)
            btn.setObjectName("PillBtn")
            btn.setCheckable(True)
            btn.setMinimumWidth(0)
            btn.setCursor(Qt.CursorShape.PointingHandCursor)
            btn.setSizePolicy(QSizePolicy.Policy.Ignored, QSizePolicy.Policy.Fixed)
            btn.clicked.connect(lambda _checked, oid=opt_id: self.changed.emit(oid))
            self._group.addButton(btn)
            self._layout.addWidget(btn, idx // cols, idx % cols)
            self._buttons[opt_id] = btn
            self._id_order.append(opt_id)

        if selected_id and selected_id in self._buttons:
            self._buttons[selected_id].setChecked(True)
        elif self._id_order:
            self._buttons[self._id_order[0]].setChecked(True)

    def set_selected(self, option_id: str) -> None:
        btn = self._buttons.get(option_id)
        if btn is not None:
            btn.blockSignals(True)
            btn.setChecked(True)
            btn.blockSignals(False)

    def current_id(self) -> str:
        for oid in self._id_order:
            if self._buttons[oid].isChecked():

class NeonMeterBar(QWidget):
    """Custom QPainter load/memory meter with 88% amber threshold and pulsing >100% overflow."""

    def __init__(self, title: str, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self._title = title
        self._right_text = "0%"
        self._ratio = 0.0
        self._base_color = QColor(CYAN)
        self._is_overflow = False
        self._pulse_phase = 0.0
        self._pulse_dir = 1

        self.setMinimumHeight(38)
        self.setMaximumHeight(42)

        self._pulse_timer = QTimer(self)
        self._pulse_timer.setInterval(45)
        self._pulse_timer.timeout.connect(self._tick_pulse)

    def _tick_pulse(self) -> None:
        self._pulse_phase += 0.09 * self._pulse_dir
        if self._pulse_phase >= 1.0:
            self._pulse_phase = 1.0
            self._pulse_dir = -1
        elif self._pulse_phase <= 0.0:
            self._pulse_phase = 0.0
            self._pulse_dir = 1
        self.update()

    def set_percentage(self, title: str, pct: float, accent_hex: str) -> None:
        self._title = title
        self._ratio = max(0.0, pct / 100.0)
        self._right_text = f"{pct:.0f}%"
        self._is_overflow = pct > 100.0
        if pct >= 95.0:
            self._base_color = QColor(MAGENTA)
        elif pct >= 85.0:
            self._base_color = QColor(AMBER)
        else:
            self._base_color = QColor(accent_hex)
        self._sync_pulse_timer()
        self.update()

    def set_memory(
        self,
        title: str,
        used_gb: float,
        total_gb: float,
        overflow_gb: float,
    ) -> None:
        self._title = title
        ratio = used_gb / max(0.1, total_gb)
        self._ratio = ratio
        self._is_overflow = overflow_gb > 0.0
        if self._is_overflow:
            self._right_text = f"{used_gb:.1f} / {total_gb:.0f} GB (+{overflow_gb:.1f} SPILL!)"
            self._base_color = QColor(CRIMSON)
        elif ratio >= 0.88:
            self._right_text = f"{used_gb:.1f} / {total_gb:.0f} GB ({ratio * 100:.0f}%)"
            self._base_color = QColor(AMBER)
        else:
            self._right_text = f"{used_gb:.1f} / {total_gb:.0f} GB ({ratio * 100:.0f}%)"
            self._base_color = QColor(EMERALD if ratio < 0.72 else CYAN)
        self._sync_pulse_timer()
        self.update()

    def _sync_pulse_timer(self) -> None:
        if self._is_overflow:
            if not self._pulse_timer.isActive():
                self._pulse_timer.start()
        else:
            if self._pulse_timer.isActive():
                self._pulse_timer.stop()
            self._pulse_phase = 0.0

    def paintEvent(self, _event: QPaintEvent) -> None:
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)

        w = float(self.width())
        label_font = QFont("Segoe UI", 9)
        label_font.setBold(True)
        painter.setFont(label_font)

        painter.setPen(QColor(TEXT_PRIMARY))
        painter.drawText(QRectF(0, 0, w * 0.55, 18), Qt.AlignmentFlag.AlignLeft | Qt.AlignmentFlag.AlignVCenter, self._title)

        right_color = QColor(CRIMSON if self._is_overflow else TEXT_SECONDARY)
        painter.setPen(right_color)
        painter.drawText(QRectF(w * 0.35, 0, w * 0.65, 18), Qt.AlignmentFlag.AlignRight | Qt.AlignmentFlag.AlignVCenter, self._right_text)

        bar_rect = QRectF(0, 21, w, 12)
        track_color = QColor(255, 255, 255, 20)
        painter.setPen(QPen(QColor(255, 255, 255, 34), 1.0))
        painter.setBrush(track_color)
        painter.drawRoundedRect(bar_rect, 6, 6)

        clamped = min(1.0, max(0.02, self._ratio))
        fill_w = max(8.0, (w - 2.0) * clamped)
        fill_rect = QRectF(1, 22, fill_w, 10)

        if self._is_overflow:
            alpha = int(175 + 80 * self._pulse_phase)
            c1 = QColor(255, 46, 147, alpha)
            c2 = QColor(255, 77, 109, alpha)
            glow_pen = QPen(QColor(255, 77, 109, int(120 + 130 * self._pulse_phase)), 1.5)
            painter.setPen(glow_pen)
            painter.drawRoundedRect(bar_rect, 6, 6)
        else:
            c1 = QColor(self._base_color)
            c2 = QColor(self._base_color).lighter(125)
            c1.setAlpha(210)
            c2.setAlpha(245)

        grad = QLinearGradient(fill_rect.topLeft(), fill_rect.topRight())
        grad.setColorAt(0.0, c1)
        grad.setColorAt(1.0, c2)
        painter.setPen(Qt.PenStyle.NoPen)
        painter.setBrush(grad)
        painter.drawRoundedRect(fill_rect, 5, 5)

        # Subtle 88% marker line
        marker_x = w * 0.88
        painter.setPen(QPen(QColor(255, 255, 255, 70), 1.0, Qt.PenStyle.DotLine))
        painter.drawLine(QPointF(marker_x, 21), QPointF(marker_x, 33))

class MultiGPUComparisonChart(QWidget):
    """Live QPainter horizontal bar chart comparing all GPUs + PS5 / PS5 Pro console baselines."""

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)
        self._entries: list[GPUComparisonEntry] = []
        self._bar_rects: list[tuple[QRectF, GPUComparisonEntry]] = []
        self.setMouseTracking(True)
        self.setMinimumHeight(330)

    def set_entries(self, entries: list[GPUComparisonEntry]) -> None:
        self._entries = entries
        needed_h = max(310, 34 + len(entries) * 26)
        self.setMinimumHeight(needed_h)
        self.update()

    def mouseMoveEvent(self, event) -> None:
        pos = event.position()
        for rect, entry in self._bar_rects:
            if rect.contains(pos):
                spill = " | VRAM OVERFLOW!" if entry.has_vram_overflow else ""
                QToolTip.showText(
                    event.globalPosition().toPoint(),
                    f"<b>{entry.label}</b><br/>"
                    f"Base Avg: {entry.base_fps:.1f} FPS | 1% Low: {entry.low_1pct_fps:.1f} FPS{spill}<br/>"
                    f"<span style='color:#94A3B8'>{entry.notes}</span>",
                    self,
                )
                return
        QToolTip.hideText()
        super().mouseMoveEvent(event)

    def paintEvent(self, _event: QPaintEvent) -> None:
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)

        self._bar_rects.clear()
        if not self._entries:
            return

        w = float(self.width())
        label_col_w = min(102.0, max(76.0, w * 0.29))
        right_pad = 38.0
        chart_left = label_col_w
        chart_w = max(80.0, w - chart_left - right_pad)
        top_offset = 24.0
        row_h = 25.0

        max_fps = max([75.0] + [e.base_fps for e in self._entries]) * 1.08

        # Header reference lines (30 FPS PS5 Target & 60 FPS Smooth Target)
        ref_font = QFont("Segoe UI", 8)
        ref_font.setBold(True)
        painter.setFont(ref_font)

        chart_bottom = top_offset + len(self._entries) * row_h
        for target_fps, label, color_hex in (
            (30.0, "30 FPS", AMBER),
            (60.0, "60 FPS", EMERALD),
        ):
            if target_fps <= max_fps:
                rx = chart_left + (target_fps / max_fps) * chart_w
                c = QColor(color_hex)
                c.setAlpha(155)
                pen = QPen(c, 1.2, Qt.PenStyle.DashLine)
                painter.setPen(pen)
                painter.drawLine(QPointF(rx, top_offset - 4), QPointF(rx, chart_bottom))

                painter.setPen(QColor(color_hex))
                lbl_x = max(chart_left - 10.0, min(w - 52.0, rx - 26.0))
                painter.drawText(
                    QRectF(lbl_x, 0, 52, 18),
                    Qt.AlignmentFlag.AlignCenter,
                    label,
                )

        name_font = QFont("Segoe UI", 8)
        val_font = QFont("Segoe UI", 8)
        val_font.setBold(True)

        for idx, entry in enumerate(self._entries):
            y = top_offset + idx * row_h
            row_rect = QRectF(0, y, w, row_h - 2)
            self._bar_rects.append((row_rect, entry))

            if entry.is_selected:
                painter.setPen(QPen(QColor(0, 240, 255, 130), 1.0))
                painter.setBrush(QColor(0, 240, 255, 24))
                painter.drawRoundedRect(row_rect, 5, 5)

            # GPU name
            f = QFont(name_font)
            f.setBold(entry.is_selected or entry.is_reference_console)
            painter.setFont(f)
            if entry.is_selected:
                painter.setPen(QColor(CYAN))
            elif entry.is_reference_console:
                painter.setPen(QColor(VIOLET))
            else:
                painter.setPen(QColor(TEXT_PRIMARY))

            painter.drawText(
                QRectF(4, y, label_col_w - 8, row_h - 2),
                Qt.AlignmentFlag.AlignRight | Qt.AlignmentFlag.AlignVCenter,
                entry.label,
            )

            # Bar track
            bar_y = y + 5
            bar_h = row_h - 12
            painter.setPen(Qt.PenStyle.NoPen)
            painter.setBrush(QColor(255, 255, 255, 14))
            painter.drawRoundedRect(QRectF(chart_left, bar_y, chart_w, bar_h), 4, 4)

            # Bar fill
            bar_w = max(4.0, min(chart_w, (entry.base_fps / max_fps) * chart_w))
            fill_rect = QRectF(chart_left, bar_y, bar_w, bar_h)

            if entry.is_reference_console:
                grad = QLinearGradient(fill_rect.topLeft(), fill_rect.topRight())
                grad.setColorAt(0.0, QColor(124, 58, 237, 155))
                grad.setColorAt(1.0, QColor(167, 139, 250, 210))
            elif entry.is_selected:
                grad = QLinearGradient(fill_rect.topLeft(), fill_rect.topRight())
                grad.setColorAt(0.0, QColor(CYAN))
                grad.setColorAt(1.0, QColor(MAGENTA))
            elif entry.has_vram_overflow:
                grad = QLinearGradient(fill_rect.topLeft(), fill_rect.topRight())
                grad.setColorAt(0.0, QColor(255, 77, 109, 175))
                grad.setColorAt(1.0, QColor(255, 184, 0, 200))
            else:
                accent = QColor(fps_accent_color(entry.base_fps))
                accent.setAlpha(185)
                grad = QLinearGradient(fill_rect.topLeft(), fill_rect.topRight())
                grad.setColorAt(0.0, accent.darker(125))
                grad.setColorAt(1.0, accent)

            painter.setBrush(grad)
            painter.drawRoundedRect(fill_rect, 4, 4)

            # Value label
            painter.setFont(val_font)
            val_color = QColor(CRIMSON if entry.has_vram_overflow else TEXT_PRIMARY)
            painter.setPen(val_color)
            suffix = " !" if entry.has_vram_overflow else ""
            painter.drawText(
                QRectF(chart_left + bar_w + 5, y, right_pad - 6, row_h - 2),
                Qt.AlignmentFlag.AlignLeft | Qt.AlignmentFlag.AlignVCenter,
                f"{entry.base_fps:.0f}{suffix}",
            )

class App(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title("ابزار اسکن و ادغام پوشه‌ها در یک فایل متنی")
        self.geometry("700x770")
        self.minsize(600, 670)

        self.root_path = Path.cwd()
        self.dir_check_vars = {}  # برای ذخیره وضعیت تیک چک‌باکس‌ها
        self.include_root_files_var = tk.BooleanVar(value=True)
        self.paths_only_var = tk.BooleanVar(value=False)  # گزینه جدید: فقط مسیر فایل‌ها

        self._build_ui()

    def _build_ui(self):
        # 1. فریم انتخاب مسیر مبدا
        frame_input = ttk.LabelFrame(self, text=" ۱. انتخاب پوشه پروژه (Root Directory) ", padding=10)
        frame_input.pack(fill="x", padx=10, pady=5)

        self.entry_root = ttk.Entry(frame_input)
        self.entry_root.insert(0, str(self.root_path))
        self.entry_root.pack(side="left", fill="x", expand=True, padx=(0, 5))

        btn_browse_root = ttk.Button(frame_input, text="انتخاب مسیر...", command=self.browse_root_dir)
        btn_browse_root.pack(side="right")

        # 2. فریم انتخاب پوشه‌ها
        frame_dirs = ttk.LabelFrame(self, text=" ۲. انتخاب پوشه‌های مورد نظر برای استخراج ", padding=10)
        frame_dirs.pack(fill="both", expand=True, padx=10, pady=5)

        # دکمه‌های کنترلی سریع
        btn_box = ttk.Frame(frame_dirs)
        btn_box.pack(fill="x", pady=(0, 5))

        ttk.Button(btn_box, text="انتخاب همه", command=self.select_all_dirs).pack(side="left", padx=2)
        ttk.Button(btn_box, text="لغو همه", command=self.deselect_all_dirs).pack(side="left", padx=2)
        ttk.Checkbutton(btn_box, text="شامل فایل‌های موجود در ریشه (Root Files)", variable=self.include_root_files_var).pack(side="right", padx=5)

        # ساخت اسکرول‌بار برای لیست پوشه‌ها
        canvas_frame = ttk.Frame(frame_dirs)
        canvas_frame.pack(fill="both", expand=True)

        self.canvas = tk.Canvas(canvas_frame, borderwidth=0, highlightthickness=0)
        self.scrollbar = ttk.Scrollbar(canvas_frame, orient="vertical", command=self.canvas.yview)
        self.scrollable_frame = ttk.Frame(self.canvas)

        self.scrollable_frame.bind(
            "<Configure>",
            lambda e: self.canvas.configure(scrollregion=self.canvas.bbox("all"))
        )
        self.canvas.create_window((0, 0), window=self.scrollable_frame, anchor="nw")
        self.canvas.configure(yscrollcommand=self.scrollbar.set)

        self.canvas.pack(side="left", fill="both", expand=True)
        self.scrollbar.pack(side="right", fill="y")

        # 3. فریم فایل خروجی و تنظیمات آن
        frame_output = ttk.LabelFrame(self, text=" ۳. فایل خروجی و حالت استخراج ", padding=10)
        frame_output.pack(fill="x", padx=10, pady=5)

        out_path_frame = ttk.Frame(frame_output)
        out_path_frame.pack(fill="x")

        self.entry_output = ttk.Entry(out_path_frame)
        self.entry_output.insert(0, str(self.root_path / "all_files.txt"))
        self.entry_output.pack(side="left", fill="x", expand=True, padx=(0, 5))

        btn_browse_out = ttk.Button(out_path_frame, text="تغییر مقصد...", command=self.browse_output_file)
        btn_browse_out.pack(side="right")

        # چک‌باکس جدید برای خروجی فقط مسیر فایل‌ها
        ttk.Checkbutton(
            frame_output,
            text="فقط استخراج مسیر کامل فایل‌ها (بدون محتوای فایل‌ها)",
            variable=self.paths_only_var
        ).pack(anchor="w", pady=(8, 0))

        # 4. دکمه اجرا
        self.btn_run = tk.Button(
            self, text="شروع اسکن و ایجاد فایل", font=("Segoe UI", 11, "bold"),
            bg="#007acc", fg="white", activebackground="#005999", activeforeground="white",
            relief="flat", pady=8, command=self.start_processing
        )
        self.btn_run.pack(fill="x", padx=10, pady=8)

        # 5. گزارش و کنسول وضعیت
        frame_logs = ttk.LabelFrame(self, text=" گزارش وضعیت و آمار ", padding=5)
        frame_logs.pack(fill="both", expand=True, padx=10, pady=(0, 10))

        self.txt_log = scrolledtext.ScrolledText(frame_logs, height=8, font=("Consolas", 9))
        self.txt_log.pack(fill="both", expand=True)

        # پر کردن اولیه لیست پوشه‌ها
        self.refresh_directories_list()

    def browse_root_dir(self):
        selected = filedialog.askdirectory(initialdir=self.root_path, title="انتخاب پوشه پروژه")
        if selected:
            self.root_path = Path(selected)
            self.entry_root.delete(0, tk.END)
            self.entry_root.insert(0, str(self.root_path))
            self.entry_output.delete(0, tk.END)
            self.entry_output.insert(0, str(self.root_path / "all_files.txt"))
            self.refresh_directories_list()

    def browse_output_file(self):
        selected = filedialog.asksaveasfilename(
            initialdir=self.root_path,
            initialfile="all_files.txt",
            defaultextension=".txt",
            filetypes=[("Text files", "*.txt"), ("All files", "*.*")]
        )
        if selected:
            self.entry_output.delete(0, tk.END)
            self.entry_output.insert(0, selected)

    def refresh_directories_list(self):
        # پاک کردن المان‌های قبلی
        for widget in self.scrollable_frame.winfo_children():
            widget.destroy()
        self.dir_check_vars.clear()

        try:
            subdirs = sorted([p for p in self.root_path.iterdir() if p.is_dir()])
        except Exception as e:
            self.log(f"خطا در خواندن پوشه: {e}")
            return

        if not subdirs:
            lbl = ttk.Label(self.scrollable_frame, text="هیچ زیرپوشه‌ای در این مسیر پیدا نشد.", foreground="gray")
            lbl.pack(anchor="w", padx=5, pady=5)
            return

        for directory in subdirs:
            dirname = directory.name
            # اگر پوشه سیستمی/بی‌اهمیت باشد پیش‌فرض غیرفعال است
            is_default_checked = dirname not in DEFAULT_SKIP_DIRS
            var = tk.BooleanVar(value=is_default_checked)
            self.dir_check_vars[dirname] = var

            chk = ttk.Checkbutton(
                self.scrollable_frame,
                text=dirname if is_default_checked else f"{dirname} (پیشنهاد حذف)",
                variable=var
            )
            chk.pack(anchor="w", padx=5, pady=2)

    def select_all_dirs(self):
        for var in self.dir_check_vars.values():
            var.set(True)

    def deselect_all_dirs(self):
        for var in self.dir_check_vars.values():
            var.set(False)

    def log(self, message):
        self.txt_log.insert(tk.END, message + "\n")
        self.txt_log.see(tk.END)

    def start_processing(self):
        # غیرفعال کردن دکمه حین پردازش
        self.btn_run.config(state="disabled")
        self.txt_log.delete(1.0, tk.END)
        # اجرای پردازش در ترد جداگانه تا UI هنگ نکند
        threading.Thread(target=self._process_files, daemon=True).start()

    def _process_files(self):
        try:
            root = Path(self.entry_root.get()).resolve()
            output_file = Path(self.entry_output.get()).resolve()

            if not root.is_dir():
                self.log("خطا: مسیر پوشه مبدا نامعتبر است!")
                messagebox.showerror("خطا", "پوشه مبدا وجود ندارد.")
                return

            selected_dirs = {name for name, var in self.dir_check_vars.items() if var.get()}
            include_root_files = self.include_root_files_var.get()
            paths_only = self.paths_only_var.get()

            self.log(f"شروع پردازش از: {root}")
            self.log(f"پوشه‌های انتخاب‌شده: {', '.join(selected_dirs) if selected_dirs else 'هیچ‌کدام'}")
            self.log(f"فایل‌های ریشه: {'بله' if include_root_files else 'خیر'}")
            self.log(f"حالت استخراج: {'فقط مسیر فایل‌ها' if paths_only else 'مسیر + محتوای فایل‌ها'}")
            self.log("-" * 50)

            files_to_process = []

            # 1. فایل‌های مستقیم ریشه
            if include_root_files:
                for item in root.iterdir():
                    if item.is_file():
                        if item.name in DEFAULT_SKIP_FILES:
                            continue
                        if item.suffix.lower() in DEFAULT_SKIP_EXTS:
                            continue
                        if item.resolve() == output_file:
                            continue
                        files_to_process.append(item)

            # 2. اسکن زیرپوشه‌های انتخاب‌شده
            for dirname in selected_dirs:
                target_dir = root / dirname
                if not target_dir.exists():
                    continue

                for file_path in target_dir.rglob("*"):
                    relative_parts = file_path.relative_to(root).parts

                    # صرف نظر از پوشه‌های غیرمجاز درون مسیر
                    if any(part in DEFAULT_SKIP_DIRS for part in relative_parts):
                        continue

                    if file_path.is_file():
                        if file_path.name in DEFAULT_SKIP_FILES:
                            continue
                        if file_path.suffix.lower() in DEFAULT_SKIP_EXTS:
                            continue
                        if file_path.resolve() == output_file:
                            continue
                        files_to_process.append(file_path)

            # مرتب‌سازی مسیرها
            files_to_process.sort(key=lambda f: f.relative_to(root).as_posix())

            total_found = len(files_to_process)
            self.log(f"تعداد کل فایل‌های تایید شده: {total_found}")

            if total_found == 0:
                self.log("هیچ فایلی مطابق معیارهای انتخابی پیدا نشد.")
                messagebox.showwarning("هشدار", "هیچ فایلی برای کپی پیدا نشد!")
                return

            read_errors = 0

            if paths_only:
                # حالت فقط مسیر کامل فایل‌ها
                lines = [str(filepath.resolve()) for filepath in files_to_process]
            else:
                # حالت معمول: هدر + محتوای فایل‌ها
                lines = [
                    "=" * 80,
                    f"  Generated from: {root}",
                    f"  Total Files Included: {total_found}",
                    "=" * 80,
                    "",
                ]

                for idx, filepath in enumerate(files_to_process, 1):
                    rel_str = filepath.relative_to(root).as_posix()
                    lines.append("=" * 80)
                    lines.append(f"FILE: {rel_str}")
                    lines.append("=" * 80)

                    try:
                        # تلاش برای خواندن با utf-8
                        content = filepath.read_text(encoding="utf-8", errors="replace")
                        lines.append(content)
                    except Exception as e:
                        read_errors += 1
                        lines.append(f"[ERROR reading file: {e}]")

                    lines.append("")

            # نوشتن در فایل خروجی
            final_content = "\n".join(lines)
            output_file.write_text(final_content, encoding="utf-8")

            file_size_kb = output_file.stat().st_size / 1024

            self.log("-" * 50)
            self.log(f"عملیات با موفقیت انجام شد!")
            self.log(f"✔ فایل ذخیره شد در: {output_file}")
            self.log(f"✔ تعداد کل فایل‌ها: {total_found}")
            self.log(f"✔ حجم فایل خروجی: {file_size_kb:.1f} KB")
            if read_errors > 0:
                self.log(f"⚠ تعداد خطاهای خواندن: {read_errors}")

            messagebox.showinfo(
                "اتمام کار",
                f"عملیات با موفقیت پایان یافت!\n\n"
                f"تعداد فایل‌ها: {total_found}\n"
                f"حجم خروجی: {file_size_kb:.1f} KB\n"
                f"مسیر: {output_file.name}"
            )

        except Exception as e:
            self.log(f"خطای سیستمی رخ داد: {e}")
            messagebox.showerror("خطا", str(e))
        finally:
            self.btn_run.config(state="normal")


if __name__ == "__main__":
    app = App()
    app.mainloop()
